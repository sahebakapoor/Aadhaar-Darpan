#include <stdio.h>

#define MAX 10

void warshall(int graph[MAX][MAX], int n) {
    int reach[MAX][MAX];

    // Copy graph
    for(int i = 0; i < n; i++) {
        for(int j = 0; j < n; j++) {
            reach[i][j] = graph[i][j];
        }
    }

    // Warshall Algorithm
    for(int k = 0; k < n; k++) {
        for(int i = 0; i < n; i++) {
            for(int j = 0; j < n; j++) {
                reach[i][j] = reach[i][j] || 
                              (reach[i][k] && reach[k][j]);
            }
        }
    }

    printf("Transitive Closure Matrix:\n");
    for(int i = 0; i < n; i++) {
        for(int j = 0; j < n; j++) {
            printf("%d ", reach[i][j]);
        }
        printf("\n");
    }
}

int main() {
    int n, graph[MAX][MAX];

    printf("Enter number of vertices: ");
    scanf("%d", &n);

    printf("Enter adjacency matrix:\n");
    for(int i = 0; i < n; i++) {
        for(int j = 0; j < n; j++) {
            scanf("%d", &graph[i][j]);
        }
    }

    warshall(graph, n);

    return 0;
}
