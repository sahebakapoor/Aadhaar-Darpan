#include <stdio.h>
#include <stdlib.h>

#define MAX 10

struct edge {
    int u, v, w;
};

int parent[MAX];

// Find (Union-Find)
int find(int i) {
    while(parent[i] != i)
        i = parent[i];
    return i;
}

// Union
void unionSet(int i, int j) {
    parent[i] = j;
}

// Sort edges by weight
void sortEdges(struct edge e[], int m) {
    for(int i = 0; i < m - 1; i++) {
        for(int j = 0; j < m - i - 1; j++) {
            if(e[j].w > e[j + 1].w) {
                struct edge temp = e[j];
                e[j] = e[j + 1];
                e[j + 1] = temp;
            }
        }
    }
}

int main() {
    int n, m;

    printf("Enter number of vertices and edges: ");
    scanf("%d %d", &n, &m);

    struct edge e[m];

    printf("Enter edges (u v weight):\n");
    for(int i = 0; i < m; i++) {
        scanf("%d %d %d", &e[i].u, &e[i].v, &e[i].w);
    }

    sortEdges(e, m);

    for(int i = 0; i < n; i++)
        parent[i] = i;

    printf("Edges in MST:\n");

    for(int i = 0; i < m; i++) {
        int u = find(e[i].u);
        int v = find(e[i].v);

        if(u != v) {
            printf("%d - %d : %d\n", e[i].u, e[i].v, e[i].w);
            unionSet(u, v);
        }
    }

    return 0;
}
