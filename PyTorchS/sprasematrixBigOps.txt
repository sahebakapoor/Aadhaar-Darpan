#include <stdio.h>

int main() {
    int n, i, j;

    printf("Enter order of square matrix (n x n): ");
    scanf("%d", &n);

    int x[n][n];

    printf("Enter matrix elements:\n");
    for(i = 0; i < n; i++) {
        for(j = 0; j < n; j++) {
            scanf("%d", &x[i][j]);
        }
    }

    // a) Transpose
    printf("\nTranspose of matrix:\n");
    for(i = 0; i < n; i++) {
        for(j = 0; j < n; j++) {
            printf("%d ", x[j][i]);
        }
        printf("\n");
    }

    // b) Row-wise sum
    printf("\nRow-wise sum:\n");
    for(i = 0; i < n; i++) {
        int sum = 0;
        for(j = 0; j < n; j++) {
            sum += x[i][j];
        }
        printf("Row %d sum = %d\n", i + 1, sum);
    }

    // c) Column-wise sum
    printf("\nColumn-wise sum:\n");
    for(j = 0; j < n; j++) {
        int sum = 0;
        for(i = 0; i < n; i++) {
            sum += x[i][j];
        }
        printf("Column %d sum = %d\n", j + 1, sum);
    }

    // d) Sum of off-diagonal elements
    int offSum = 0;
    for(i = 0; i < n; i++) {
        for(j = 0; j < n; j++) {
            if(i != j) {
                offSum += x[i][j];
            }
        }
    }
    printf("\nSum of off-diagonal elements = %d\n", offSum);

    // e) Upper triangular matrix
    printf("\nUpper Triangular Matrix:\n");
    for(i = 0; i < n; i++) {
        for(j = 0; j < n; j++) {
            if(i <= j)
                printf("%d ", x[i][j]);
            else
                printf("0 ");
        }
        printf("\n");
    }

    // f) Lower triangular matrix
    printf("\nLower Triangular Matrix:\n");
    for(i = 0; i < n; i++) {
        for(j = 0; j < n; j++) {
            if(i >= j)
                printf("%d ", x[i][j]);
            else
                printf("0 ");
        }
        printf("\n");
    }

    // g) Diagonal elements only
    printf("\nDiagonal elements:\n");
    for(i = 0; i < n; i++) {
        for(j = 0; j < n; j++) {
            if(i == j)
                printf("%d ", x[i][j]);
            else
                printf("0 ");
        }
        printf("\n");
    }

    return 0;
}
