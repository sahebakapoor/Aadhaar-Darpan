#include <stdio.h>
#include <stdlib.h>

#define SIZE 10

struct node {
    int data;
    struct node* next;
};

struct node* table[SIZE];

// Hash function
int hash(int key) {
    return key % SIZE;
}

// Insert into hash table
void insert(int key) {
    int index = hash(key);

    struct node* newNode = (struct node*)malloc(sizeof(struct node));
    newNode->data = key;
    newNode->next = table[index];
    table[index] = newNode;
}

// Display hash table
void display() {
    for(int i = 0; i < SIZE; i++) {
        printf("Index %d: ", i);
        struct node* temp = table[i];

        while(temp != NULL) {
            printf("%d -> ", temp->data);
            temp = temp->next;
        }
        printf("NULL\n");
    }
}

int main() {
    int arr[20];

    printf("Enter 20 integers:\n");
    for(int i = 0; i < 20; i++) {
        scanf("%d", &arr[i]);
    }

    // Initialize table
    for(int i = 0; i < SIZE; i++)
        table[i] = NULL;

    // Insert elements
    for(int i = 0; i < 20; i++) {
        insert(arr[i]);
    }

    printf("\nHash Table (Separate Chaining):\n");
    display();

    return 0;
}
