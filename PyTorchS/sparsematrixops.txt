#include <stdio.h>

int main() {
    int m, n, i, j;
    int count = 0;

    printf("Enter number of rows and columns: ");
    scanf("%d %d", &m, &n);

    int a[m][n];

    printf("Enter matrix elements:\n");
    for(i = 0; i < m; i++) {
        for(j = 0; j < n; j++) {
            scanf("%d", &a[i][j]);
            if(a[i][j] == 0)
                count++;
        }
    }

    // Check sparse condition
    if(count > (m * n) / 2) {
        printf("\nMatrix is a Sparse Matrix.\n");

        // Count non-zero elements
        int nonZero = m * n - count;

        // Sparse representation: row, col, value
        int sparse[nonZero + 1][3];

        // First row = metadata
        sparse[0][0] = m;
        sparse[0][1] = n;
        sparse[0][2] = nonZero;

        int k = 1;

        for(i = 0; i < m; i++) {
            for(j = 0; j < n; j++) {
                if(a[i][j] != 0) {
                    sparse[k][0] = i;
                    sparse[k][1] = j;
                    sparse[k][2] = a[i][j];
                    k++;
                }
            }
        }

        // Display sparse matrix
        printf("\nSparse Matrix Representation (Row Col Value):\n");
        for(i = 0; i < nonZero + 1; i++) {
            printf("%d %d %d\n", sparse[i][0], sparse[i][1], sparse[i][2]);
        }
    }
    else {
        printf("\nMatrix is NOT a Sparse Matrix.\n");
    }

    return 0;
}
